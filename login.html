// TODO(backend): point data-auth-endpoint at the auth service once it exists
// (Supabase auth / app_auth). Until then, sign-in shows a pending message and
// guest access is the entry point.
document.getElementById('portal-login').addEventListener('submit', function (e) {
  var endpoint = this.getAttribute('data-auth-endpoint');
  if (!endpoint) {
    e.preventDefault();
    var msg = document.getElementById('login-msg');
    msg.textContent = 'Sign-in is being wired to the backend — continue as guest for now.';
    msg.hidden = false;
  }
});
